# ===== KNIGHT CLASS =====
class Knight:
    def __init__(self, name, damage):
        self.__name = name
        self.__damage = damage

    def get_name(self):
        return self.__name

    def get_damage(self):
        return self.__damage

    def set_damage(self, new_damage):
        if new_damage > 0:
          self.__damage = new_damage

    def attack(self):
        print(f"{self.get_name()} swings his sword!")
        return self.__damage
